  // Open / close modals

    e.preventDefault();
    const m = document.getElementById(el.dataset.open);
    m.classList.add('open'); m.setAttribute('aria-hidden', 'false');
  }));
  const closeAll = () => document.querySelectorAll('.overlay.open').forEach(m => {
    m.classList.remove('open'); m.setAttribute('aria-hidden', 'true');
  });
  document.querySelectorAll('[data-close]').forEach(b => b.addEventListener('click', closeAll));
  document.querySelectorAll('.overlay').forEach(o => o.addEventListener('click', e => { if (e.target === o) closeAll(); }));
